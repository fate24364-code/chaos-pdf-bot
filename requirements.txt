import os
import logging
import sqlite3
from contextlib import closing

from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import (
    Application, CommandHandler, CallbackQueryHandler,
    MessageHandler, ContextTypes, filters,
)

TOKEN = os.environ.get("BOT_TOKEN", "").strip()
ADMIN_ID = 8481208591
DB_PATH = os.environ.get("DB_PATH", "novels.db")

logging.basicConfig(
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    level=logging.INFO,
)
logger = logging.getLogger(__name__)


def connect_db():
    conn = sqlite3.connect(DB_PATH, timeout=20)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    with closing(connect_db()) as db:
        db.execute("""
            CREATE TABLE IF NOT EXISTS novels (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                file_id TEXT NOT NULL,
                channels TEXT DEFAULT ''
            )
        """)
        db.commit()


def is_admin(user_id):
    return user_id == ADMIN_ID


def parse_channels(raw):
    raw = (raw or "").strip()
    if not raw or raw.lower() in {"ندارد", "هیچ", "none", "no"}:
        return []
    # Accept one channel per line, comma-separated values, or a single @username.
    parts = raw.replace(",", "\n").splitlines()
    channels = []
    for part in parts:
        channel = part.strip()
        if not channel:
            continue
        if channel.startswith("https://t.me/"):
            channel = channel.removeprefix("https://t.me/").strip("/")
        elif channel.startswith("http://t.me/"):
            channel = channel.removeprefix("http://t.me/").strip("/")
        if channel.startswith("@"):
            channel = channel[1:]
        # Numeric IDs such as -1001234567890 are accepted as-is.
        if channel.startswith("-100") and channel[4:].isdigit():
            normalized = channel
        elif channel and channel.replace("_", "").isalnum():
            normalized = "@" + channel
        else:
            normalized = channel
        if normalized not in channels:
            channels.append(normalized)
    return channels


def channel_url(channel):
    if channel.startswith("@"):
        return "https://t.me/" + channel[1:]
    if channel.startswith("https://t.me/"):
        return channel
    if channel.startswith("-100"):
        return None  # private channel IDs are not invite links
    return "https://t.me/" + channel.lstrip("/")


def get_novel(novel_id):
    with closing(connect_db()) as db:
        return db.execute(
            "SELECT id, title, file_id, channels FROM novels WHERE id=?",
            (novel_id,),
        ).fetchone()


async def safe_reply(update, text, **kwargs):
    message = update.effective_message
    if message:
        await message.reply_text(text, **kwargs)


async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if context.args and context.args[0].startswith("novel_"):
        novel_id = context.args[0].removeprefix("novel_")
        if not novel_id.isdigit():
            await safe_reply(update, "لینک دریافت رمان معتبر نیست.")
            return
        await send_novel(update, context, int(novel_id))
        return

    keyboard = []
    if is_admin(update.effective_user.id):
        keyboard.append([InlineKeyboardButton("⚙️ پنل مدیریت", callback_data="admin")])
    await safe_reply(
        update,
        "سلام! 📚\nبرای دریافت رمان از لینک اختصاصی همان رمان وارد شو.",
        reply_markup=InlineKeyboardMarkup(keyboard) if keyboard else None,
    )


async def send_novel(update, context, novel_id):
    row = get_novel(novel_id)
    if not row:
        await safe_reply(update, "این رمان پیدا نشد یا حذف شده است.")
        return

    channels = [c for c in (row["channels"] or "").split("\n") if c.strip()]
    if not channels:
        await context.bot.send_document(
            chat_id=update.effective_chat.id,
            document=row["file_id"],
            caption=f"📚 {row['title']}",
        )
        return

    buttons = []
    for channel in channels:
        url = channel_url(channel)
        if url:
            buttons.append([InlineKeyboardButton("📢 عضویت در کانال", url=url)])
        else:
            buttons.append([
                InlineKeyboardButton(
                    "📢 کانال خصوصی — لینک دعوت را از مدیر بگیر",
                    callback_data="private_channel_info",
                )
            ])
    buttons.append([
        InlineKeyboardButton("✅ عضو شدم؛ بررسی کن", callback_data=f"check_{novel_id}")
    ])
    await safe_reply(
        update,
        f"برای دریافت «{row['title']}» ابتدا در کانال(های) زیر عضو شو.",
        reply_markup=InlineKeyboardMarkup(buttons),
    )


async def check_membership(update, context, novel_id):
    query = update.callback_query
    await query.answer()
    row = get_novel(novel_id)
    if not row:
        await query.message.reply_text("این رمان پیدا نشد یا حذف شده است.")
        return

    channels = [c for c in (row["channels"] or "").split("\n") if c.strip()]
    for channel in channels:
        try:
            member = await context.bot.get_chat_member(
                chat_id=channel,
                user_id=query.from_user.id,
            )
            if member.status in ("left", "kicked", "banned"):
                await query.message.reply_text(
                    f"هنوز عضو {channel} نشدی. بعد از عضویت دوباره بررسی کن."
                )
                return
        except Exception:
            logger.exception("Membership check failed for channel %s", channel)
            await query.message.reply_text(
                "نتونستم عضویت این کانال رو بررسی کنم. مدیر باید مطمئن بشه ربات "
                "در کانال ادمینه و شناسه/نام کانال درست وارد شده. برای کانال خصوصی، "
                "شناسهٔ عددی کانال (مثل -100...) را وارد کن و لینک دعوت را جداگانه "
                "در تنظیمات کانال به کاربران بده."
            )
            return

    await context.bot.send_document(
        chat_id=query.message.chat_id,
        document=row["file_id"],
        caption=f"📚 {row['title']}",
    )


async def admin_panel(update, context):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    keyboard = [
        [InlineKeyboardButton("➕ افزودن رمان", callback_data="add")],
        [InlineKeyboardButton("🗂️ مدیریت / حذف / ویرایش رمان‌ها", callback_data="manage")],
        [InlineKeyboardButton("❌ لغو عملیات فعلی", callback_data="cancel")],
    ]
    await query.message.reply_text(
        "⚙️ پنل مدیریت",
        reply_markup=InlineKeyboardMarkup(keyboard),
    )


async def add_novel(update, context):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    context.user_data["flow"] = {"action": "add", "stage": "title"}
    await query.message.reply_text(
        "اسم رمان رو بفرست. هر وقت پشیمون شدی /cancel بفرست."
    )


async def show_manage(update, context):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    with closing(connect_db()) as db:
        rows = db.execute("SELECT id, title FROM novels ORDER BY id DESC").fetchall()
    if not rows:
        await query.message.reply_text("هنوز رمانی ثبت نشده.")
        return
    buttons = [
        [InlineKeyboardButton(f"🗑️ {row['title']}", callback_data=f"delete_{row['id']}"),
         InlineKeyboardButton("✏️ ویرایش", callback_data=f"edit_{row['id']}")]
        for row in rows
    ]
    buttons.append([InlineKeyboardButton("⬅️ پنل مدیریت", callback_data="admin")])
    await query.message.reply_text(
        "برای حذف یا ویرایش، رمان موردنظر رو انتخاب کن:",
        reply_markup=InlineKeyboardMarkup(buttons),
    )


async def ask_delete(update, context, novel_id):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    row = get_novel(novel_id)
    if not row:
        await query.message.reply_text("این رمان پیدا نشد.")
        return
    context.user_data["flow"] = {"action": "delete", "novel_id": novel_id, "stage": "confirm"}
    keyboard = [[
        InlineKeyboardButton("بله، حذف شود", callback_data=f"confirm_delete_{novel_id}"),
        InlineKeyboardButton("نه، برگرد", callback_data="manage"),
    ]]
    await query.message.reply_text(
        f"مطمئنی می‌خوای «{row['title']}» از ربات حذف بشه؟",
        reply_markup=InlineKeyboardMarkup(keyboard),
    )


async def confirm_delete(update, context, novel_id):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    with closing(connect_db()) as db:
        db.execute("DELETE FROM novels WHERE id=?", (novel_id,))
        db.commit()
    context.user_data.pop("flow", None)
    await query.message.reply_text("🗑️ رمان از فهرست ربات حذف شد. فایل اصلی که قبلاً جایی ذخیره شده باشد حذف نمی‌شود.")


async def ask_edit(update, context, novel_id):
    query = update.callback_query
    await query.answer()
    if not is_admin(query.from_user.id):
        await query.message.reply_text("اجازه دسترسی نداری.")
        return
    row = get_novel(novel_id)
    if not row:
        await query.message.reply_text("این رمان پیدا نشد.")
        return
    context.user_data["flow"] = {"action": "edit", "novel_id": novel_id, "stage": "choice"}
    keyboard = [
        [InlineKeyboardButton("تغییر اسم", callback_data="edit_title")],
        [InlineKeyboardButton("تغییر PDF", callback_data="edit_pdf")],
        [InlineKeyboardButton("تغییر کانال‌های اجباری", callback_data="edit_channels")],
        [InlineKeyboardButton("⬅️ برگشت", callback_data="manage")],
    ]
    await query.message.reply_text(
        f"کدوم بخش «{row['title']}» رو تغییر بدیم؟",
        reply_markup=InlineKeyboardMarkup(keyboard),
    )


async def begin_edit_field(update, context, stage):
    query = update.callback_query
    await query.answer()
    flow = context.user_data.get("flow", {})
    if not is_admin(query.from_user.id) or flow.get("action") != "edit":
        await query.message.reply_text("این عملیات فعال نیست؛ از پنل مدیریت دوباره شروع کن.")
        return
    flow["stage"] = stage
    context.user_data["flow"] = flow
    prompts = {
        "title": "اسم جدید رمان رو بفرست.",
        "pdf": "فایل PDF جدید رو به‌صورت فایل بفرست.",
        "channels": "کانال‌ها رو هرکدوم در یک خط بفرست. برای حذف عضویت اجباری بنویس: ندارد",
    }
    await query.message.reply_text(prompts[stage] + "\nبرای لغو /cancel بفرست.")


async def cancel(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not is_admin(update.effective_user.id):
        await safe_reply(update, "اجازه دسترسی نداری.")
        return
    context.user_data.pop("flow", None)
    await safe_reply(update, "عملیات لغو شد. چیزی ذخیره نشد.")


async def handle_admin_message(update, context):
    if not update.effective_user or not is_admin(update.effective_user.id):
        return
    flow = context.user_data.get("flow")
    if not flow:
        return
    stage = flow.get("stage")
    action = flow.get("action")

    if action == "add":
        if stage == "title":
            if not update.message.text or not update.message.text.strip():
                await update.message.reply_text("اسم رمان رو به‌صورت متن بفرست.")
                return
            flow["title"] = update.message.text.strip()
            flow["stage"] = "pdf"
            await update.message.reply_text("حالا فایل PDF رمان رو به‌صورت فایل بفرست.")
            return
        if stage == "pdf":
            document = update.message.document
            if not document or document.mime_type != "application/pdf":
                await update.message.reply_text("لطفاً فایل PDF رو با گزینهٔ ارسال فایل بفرست.")
                return
            flow["file_id"] = document.file_id
            flow["stage"] = "channels"
            await update.message.reply_text(
                "کانال‌های اجباری رو هرکدوم در یک خط بفرست.\n"
                "برای کانال عمومی: @ChannelUsername\n"
                "برای کانال خصوصی: شناسهٔ عددی -100... (ربات باید ادمین باشد).\n"
                "اگر عضویت اجباری نمی‌خوای بنویس: ندارد"
            )
            return
        if stage == "channels":
            if not update.message.text:
                await update.message.reply_text("لطفاً کانال‌ها رو متنی بفرست یا بنویس «ندارد».")
                return
            channels = parse_channels(update.message.text)
            with closing(connect_db()) as db:
                db.execute(
                    "INSERT INTO novels (title, file_id, channels) VALUES (?, ?, ?)",
                    (flow["title"], flow["file_id"], "\n".join(channels)),
                )
                db.commit()
            context.user_data.pop("flow", None)
            await update.message.reply_text("✅ رمان ذخیره شد!")
            return

    if action == "edit":
        novel_id = flow.get("novel_id")
        if stage == "title":
            if not update.message.text or not update.message.text.strip():
                await update.message.reply_text("اسم جدید رو به‌صورت متن بفرست.")
                return
            with closing(connect_db()) as db:
                db.execute("UPDATE novels SET title=? WHERE id=?", (update.message.text.strip(), novel_id))
                db.commit()
            context.user_data.pop("flow", None)
            await update.message.reply_text("✅ اسم رمان تغییر کرد.")
            return
        if stage == "pdf":
            document = update.message.document
            if not document or document.mime_type != "application/pdf":
                await update.message.reply_text("لطفاً فایل PDF رو با گزینهٔ ارسال فایل بفرست.")
                return
            with closing(connect_db()) as db:
                db.execute("UPDATE novels SET file_id=? WHERE id=?", (document.file_id, novel_id))
                db.commit()
            context.user_data.pop("flow", None)
            await update.message.reply_text("✅ فایل PDF تغییر کرد.")
            return
        if stage == "channels":
            if not update.message.text:
                await update.message.reply_text("لطفاً کانال‌ها رو متنی بفرست.")
                return
            channels = parse_channels(update.message.text)
            with closing(connect_db()) as db:
                db.execute("UPDATE novels SET channels=? WHERE id=?", ("\n".join(channels), novel_id))
                db.commit()
            context.user_data.pop("flow", None)
            await update.message.reply_text("✅ کانال‌های اجباری تغییر کردند.")
            return


async def buttons(update, context):
    query = update.callback_query
    data = query.data or ""
    if data == "private_channel_info":
        await query.answer("برای کانال خصوصی باید شناسهٔ عددی و لینک دعوت معتبر داشته باشی.", show_alert=True)
    elif data == "admin":
        await admin_panel(update, context)
    elif data == "add":
        await add_novel(update, context)
    elif data == "manage":
        await show_manage(update, context)
    elif data == "cancel":
        if not is_admin(query.from_user.id):
            await query.answer("اجازه دسترسی نداری.", show_alert=True)
        else:
            context.user_data.pop("flow", None)
            await query.answer("لغو شد.")
            await query.message.reply_text("عملیات لغو شد.")
    elif data.startswith("delete_"):
        await ask_delete(update, context, int(data.removeprefix("delete_")))
    elif data.startswith("confirm_delete_"):
        await confirm_delete(update, context, int(data.removeprefix("confirm_delete_")))
    elif data.startswith("edit_title"):
        await begin_edit_field(update, context, "title")
    elif data.startswith("edit_pdf"):
        await begin_edit_field(update, context, "pdf")
    elif data.startswith("edit_channels"):
        await begin_edit_field(update, context, "channels")
    elif data.startswith("edit_"):
        await ask_edit(update, context, int(data.removeprefix("edit_")))
    elif data.startswith("check_"):
        await check_membership(update, context, int(data.removeprefix("check_")))
    else:
        await query.answer()


def main():
    if not TOKEN:
        raise RuntimeError("متغیر محرمانه BOT_TOKEN تنظیم نشده است.")
    init_db()
    app = Application.builder().token(TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("cancel", cancel))
    app.add_handler(CallbackQueryHandler(buttons))
    app.add_handler(MessageHandler(filters.ALL & ~filters.COMMAND, handle_admin_message))
    app.run_polling()


if __name__ == "__main__":
    main()
